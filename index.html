World Flags Explorer

Select a country from the dropdown below to display its flag.

regionNames = new Intl.DisplayNames(["en"], { type: "region" })

ukNations = ({
  "gb-eng": "England",
  "gb-nir": "Northern Ireland",
  "gb-sct": "Scotland",
  "gb-wls": "Wales"
})

// Combine country names with their flag features
countries = transpose(flag_features)
  .map(d => ({
    ...d,
    name: ukNations[d.code] ?? regionNames.of(d.code.toUpperCase())
  }))
  .sort((a, b) => a.name.localeCompare(b.name))

// Create an interactive dropdown menu; `viewof` exposes the selected value
viewof selected = Inputs.select(countries, {
  label: "Choose a country:",
  format: d => d.name
})

// Find other flags sharing the same colors, pattern, or symbols
similarColor = countries
  .filter(d => d.code !== selected.code && d.colors !== "Unknown" && d.colors === selected.colors)
  .slice(0, 8)

similarPattern = countries
  .filter(d => d.code !== selected.code && d.pattern !== "Unknown" && d.pattern === selected.pattern)
  .slice(0, 8)

similarSymbols = countries
  .filter(d => d.code !== selected.code && d.symbols !== "Unknown" && d.symbols !== "None" && d.symbols === selected.symbols)
  .slice(0, 8)

// Helper to render a list of matching country names
formatMatches = list => list.length > 0 ? list.map(d => d.name).join(", ") : "None found in dataset"

// Helper to render flag thumbnail images for a list of countries
renderFlagGrid = list => list.length > 0
  ? html`<div style="display: flex; flex-wrap: wrap; gap: 0.75rem; margin-bottom: 1.25rem;">
      ${list.map(d => html`<div style="text-align: center; width: 110px;">
        <img 
          src="w2560/${d.code}.png" 
          alt="Flag of ${d.name}" 
          style="width: 100px; height: 66px; object-fit: cover; border: 1px solid #ccc; border-radius: 4px;"
        />
        <div style="font-size: 0.8rem; margin-top: 0.25rem;">${d.name}</div>
      </div>`)}
    </div>`
  : html`<p style="color: #6c757d; font-size: 0.9rem;">None found in dataset</p>`

// Display the local flag image, the similarity info box, and the similar flag images underneath
html`<div style="margin-top: 1.5rem;">
  <h3>${selected.name}</h3>
  <img 
    src="w2560/${selected.code}.png" 
    alt="Flag of ${selected.name}" 
    style="width: 320px; border: 1px solid #ccc; border-radius: 4px;"
  />

  <div style="margin-top: 1.25rem; padding: 1rem 1.25rem; background: #f8f9fa; border: 1px solid #dee2e6; border-radius: 8px; max-width: 600px;">
    <h4 style="margin-top: 0;">Flag Features & Similar Flags</h4>
    <p>
      <strong>Colors (${selected.colors}):</strong><br/>
      Similar: ${formatMatches(similarColor)}
    </p>
    <p>
      <strong>Pattern (${selected.pattern}):</strong><br/>
      Similar: ${formatMatches(similarPattern)}
    </p>
    <p style="margin-bottom: 0;">
      <strong>Symbols (${selected.symbols}):</strong><br/>
      Similar: ${formatMatches(similarSymbols)}
    </p>
  </div>

  <div style="margin-top: 1.5rem;">
    <h5>Similar by Color</h5>
    ${renderFlagGrid(similarColor)}

    <h5>Similar by Pattern</h5>
    ${renderFlagGrid(similarPattern)}

    <h5>Similar by Symbols</h5>
    ${renderFlagGrid(similarSymbols)}
  </div>
</div>`
